        let aktifKat = 'hepsi';
        let tumMarkerlar = [];

        function mekanlariYukle() {
            const container = document.getElementById('location-list');
            container.innerHTML = '';
            
            tumMarkerlar.forEach(m => map.removeLayer(m.marker));
            tumMarkerlar = [];

            const aramaMetni = document.getElementById('search-input').value.toLowerCase();

            mekanlar.forEach(m => {
                const katUyum = aktifKat === 'hepsi' || m.kategori === aktifKat;
                const aramaUyum = m.isim.toLowerCase().includes(aramaMetni) || m.aciklama.toLowerCase().includes(aramaMetni);

                if (katUyum && aramaUyum) {
                    const marker = L.marker([m.en, m.boy], { icon: ikonOlustur(m.simge) }).addTo(map);
                    const gUrl = `https://www.google.com/maps/dir/?api=1&destination=${m.en},${m.boy}`;

                    let fotoHtml = '<div class="popup-gallery">';
                    m.fotograflar.forEach(imgUrl => {
                        fotoHtml += `<img src="${imgUrl}" alt="Urbex">`;
                    });
                    fotoHtml += '</div>';

                    const popupIcerik = `
                        <div style="color: #000; max-width: 260px;">
                            <h3 style="font-size:1.05rem;">${m.simge} ${m.isim}</h3>
                            <p style="font-size:0.8rem; color:#555; margin-top:2px;"><b>Kategori:</b> ${m.kategori}</p>
                            <p style="font-size:0.8rem; color:#555;"><b>Tehlike:</b> ${m.tehlike}</p>
                            ${fotoHtml}
                            <p style="font-size:0.8rem; margin-bottom:6px;">${m.aciklama}</p>
                            <a href="${gUrl}" target="_blank" style="display:block; text-align:center; background:#2ed573; color:#000; padding:5px; border-radius:4px; font-weight:bold; text-decoration:none; font-size:0.75rem;">Google Maps Yol Tarifi</a>
                        </div>
                    `;

                    marker.bindPopup(popupIcerik);
                    tumMarkerlar.push({ id: m.id, marker: marker });

                    const card = document.createElement('div');
                    card.className = 'card';
                    card.innerHTML = `
                        <span class="badge">${m.simge} ${m.kategori}</span>
                        <h3>${m.isim}</h3>
                        <p>⚠️ Tehlike: ${m.tehlike}</p>
                        <p style="font-size:0.75rem; color:#aaa; margin-bottom:6px;">${m.aciklama}</p>
                        <a href="${gUrl}" target="_blank" class="btn-nav">Google Maps Yol Tarifi</a>
                    `;

                    card.addEventListener('click', (e) => {
                        if(e.target.tagName !== 'A') {
                            map.flyTo([m.en, m.boy], 14, { duration: 1.5 });
                            marker.openPopup();
                        }
                    });

                    container.appendChild(card);
                }
            });
        }

        function kategoriFiltrele(kat, btn) {
            aktifKat = kat;
            document.querySelectorAll('.filter-btn').forEach(b => b.classList.remove('active'));
            btn.classList.add('active');
            mekanlariYukle();
        }

        function filtrele() {
            mekanlariYukle();
        }
